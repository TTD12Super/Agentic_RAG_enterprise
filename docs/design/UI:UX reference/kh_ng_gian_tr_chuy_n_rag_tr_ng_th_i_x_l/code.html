from fastapi import FastAPI, Header, HTTPException, Request
import hmac, hashlib

app = FastAPI()
WEBHOOK_SECRET = b"whsec_9941_sig_key"

@app.post("/webhook/enterprise-rag")
async def handle_rag_event(request: Request, x_signature: str = Header(None)):
    payload = await request.body()
    computed = hmac.new(WEBHOOK_SECRET, payload, hashlib.sha256).hexdigest()
    if not hmac.compare_digest(f"sha256={computed}", x_signature or ""):
        raise HTTPException(status_code=403, detail="Invalid signature")
    data = await request.json()
    print(f"Document approved: {data['doc_title']} by {data['approver']}")
    return {"status": "received"}
